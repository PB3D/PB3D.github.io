85 deallocate(l_v_s)

88 deallocate(b_v_sub_c)

90#if ldebug
91 deallocate(b_v_c)
92 deallocate(b_v_s)
93 deallocate(j_v_sup_int)
94
95 ! memory usage difference after deallocation
96 if (print_mem_usage) then
97 mem_diff = mem_diff - get_mem_usage()

100 end if
101#endif
