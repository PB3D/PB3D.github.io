33 !> \public Calculates the normal (\f$\cdot_n\f$) or geodesic (\f$\cdot_g\f$)
34 !! components of the plasma perturbation \f$\vec{\xi}\f$ or the
35 !! magnetic perturbation \f$\vec{Q} = \nabla \times \left(\vec{x} \times
36 !! \vec{B}\right)\f$.
37 !!
38 !! Which variables are plot depends on \c XUQ_style \cite Weyens3D :
39 !! - \c XUQ_style = 1: \f$X\f$ (supports parallel derivative)
40 !! - \c XUQ_style = 2: \f$U\f$ (supports parallel derivative)
41 !! - \c XUQ_style = 3: \f$Q_n\f$
42 !! - \c XUQ_style = 4: \f$Q_g\f$
43 !!
44 !! \f$X\f$ and \f$U\f$ support the calculation of the parallel derivative.
45 !!
46 !! For \f$Q_n\f$ and \f$Q_g\f$, the metric variables have to be provided as
47 !! well.
48 !!
49 !! The perturbation grid is assumed to have the same angular coordinates as
50 !! the equilibrium grid, and the normal coordinates correspond to either the
51 !! equilibrium grid (\c X_grid_style 1), the solution grid (\c X_grid_style
52 !! 2) or the enriched equilibrium grid (\c X_grid_style 3).
53 !!
54 !! The output grid, furthermore, has the angular part corresponding to the
55 !! equilibrium grid, and the normal part to the solution grid.
56 !!
57 !! \return ierr
58 interface calc_xuq
59 !> \public

150 ierr = 1

156 ierr = check_same_modes(mds_x,mds_sol)

159 sec_sol_loc => mds_sol%sec

161 n_sec = size(sec_x_loc,1)

163 ! set up normalized sqrt(sol_val)
164 sqrt_sol_val_norm = sqrt(sol%val(x_id))
165 if (ip(sqrt_sol_val_norm).gt.0) sqrt_sol_val_norm = - sqrt_sol_val_norm ! exploding solution, not the decaying one
166 sqrt_sol_val_norm = sqrt_sol_val_norm / abs(sqrt_sol_val_norm) ! normalize

168 ! allocate helper variables
169 allocate(jq(xuq_dims(3)))
170 allocate(s(xuq_dims(1),xuq_dims(2),xuq_dims(3)))
171 if ((xuq_style.eq.3 .or. xuq_style.eq.4)) &
172 &allocate(inv_j(xuq_dims(1),xuq_dims(2),xuq_dims(3)))

176 ! interpolate eq->sol
177 if (use_pol_flux_f) then
178 ierr = spline(grid_eq%loc_r_F,eq_1%q_saf_FD(:,0),&
179 &grid_sol%loc_r_F,jq,ord=norm_disc_prec_sol)
180 chckerr('')
181 else
182 ierr = spline(grid_eq%loc_r_F,eq_1%rot_t_FD(:,0),&
183 &grid_sol%loc_r_F,jq,ord=norm_disc_prec_sol)
184 chckerr('')
185 end if
186 do jd = 1,xuq_dims(2)
187 do id = 1,xuq_dims(1)
188 ierr = spline(grid_eq%loc_r_F,eq_2%S(id,jd,:),&
189 &grid_sol%loc_r_F,s(id,jd,:),ord=norm_disc_prec_sol)
190 chckerr('')
191 if ((xuq_style.eq.3 .or. xuq_style.eq.4)) then
192 ierr = spline(grid_eq%loc_r_F,&
193 &1._dp/eq_2%jac_FD(id,jd,:,0,0,0),grid_sol%loc_r_F,&
194 &inv_j(id,jd,:),ord=norm_disc_prec_sol)
195 chckerr('')
196 end if
197 end do
198 end do

200 ! initialize XUQ
201 xuq = 0._dp
202
203 ! set up normal derivative of X vec
204 ! Note: In X_style 2 (fast), the mode indices vary as a function as the
205 ! normal coordinate. All of these contiguous ranges are tabulated in
206 ! variable 'sec'.
207 if (xuq_style.eq.2 .or. xuq_style.eq.4) then
208 ! set up variables
209 allocate(sol_vec_tot(get_n_mod_tot(mds_sol),grid_sol%loc_n_r,0:1))
210 sol_vec_tot = 0._dp
211
212 ! convert to total solution vector and derivate
213 do id = 0,1
214 ierr = calc_tot_sol_vec(mds_sol,grid_sol,sol%vec(:,:,x_id),&
215 &sol_vec,deriv=id)
216 chckerr('')
217 sol_vec_tot(:,:,id) = sol_vec

220#if ldebug
221 if (debug_calc_xuq_arr) then
222 do m = 1,n_sec
223 m_tot = sec_sol_loc(m,1)-minval(sec_sol_loc(:,1))+1
224 call writo('For mode '//trim(i2str(sec_sol_loc(m,1)))//', testing &
225 &whether Dsol_vec is correct by comparing its integral &
226 &with original sol_vec')
227 kdl_sol = sec_sol_loc(m,2:3)
228 allocate(sol_vec_alt(1:kdl_sol(2)-kdl_sol(1)+1))
229 sol_vec_alt = 0._dp
230 ierr = calc_int(rp(sol_vec_tot(m_tot,kdl_sol(1):kdl_sol(2),1)),&
231 &grid_sol%loc_r_F(kdl_sol(1):kdl_sol(2)),sol_vec_alt)
232 chckerr('')
233 call print_ex_2d(['TEST_RE_Dsol_vec'],'',reshape(&
234 &[rp(sol_vec_tot(m_tot,kdl_sol(1):kdl_sol(2),0)),&
235 &sol_vec_alt],[kdl_sol(2)-kdl_sol(1)+1,2]),x=reshape(&
236 &grid_sol%loc_r_F(kdl_sol(1):kdl_sol(2)),&
237 &[kdl_sol(2)-kdl_sol(1)+1,1])/max_flux_f*2*pi)
238 ierr = calc_int(ip(sol_vec_tot(m_tot,kdl_sol(1):kdl_sol(2),1)),&
239 &grid_sol%loc_r_F(kdl_sol(1):kdl_sol(2)),sol_vec_alt)
240 chckerr('')
241 call print_ex_2d(['TEST_IM_Dsol_vec'],'',reshape(&
242 &[ip(sol_vec_tot(m_tot,kdl_sol(1):kdl_sol(2),0)),&
243 &sol_vec_alt],[kdl_sol(2)-kdl_sol(1)+1,2]),x=reshape(&
244 &grid_sol%loc_r_F(kdl_sol(1):kdl_sol(2)),&
245 &[kdl_sol(2)-kdl_sol(1)+1,1])/max_flux_f*2*pi)
246 deallocate(sol_vec_alt)
247 end do
248 end if
249#endif
250
251 ! convert back to local solution vector
252 ierr = calc_loc_sol_vec(mds_sol,grid_sol%i_min,sol_vec_tot(:,:,1),&
253 &dsol_vec)
254 chckerr('')
255
256 ! clean up
257 deallocate(sol_vec,sol_vec_tot)
258 else
259 dsol_vec = 0._dp
260 end if

262 ! interpolate or point angles
263 select case (x_grid_style)
264 case (1,3) ! equilibrium or enriched
265 allocate(theta_f(xuq_dims(1),xuq_dims(2),xuq_dims(3)))
266 allocate(zeta_f(xuq_dims(1),xuq_dims(2),xuq_dims(3)))
267 do jd = 1,xuq_dims(2)
268 do id = 1,xuq_dims(1)
269 ierr = spline(grid_eq%loc_r_F,grid_eq%theta_F(id,jd,:),&
270 &grid_sol%loc_r_F,theta_f(id,jd,:))
271 chckerr('')
272 ierr = spline(grid_eq%loc_r_F,grid_eq%zeta_F(id,jd,:),&
273 &grid_sol%loc_r_F,zeta_f(id,jd,:))
274 chckerr('')
275 end do
276 end do
277 case (2) ! solution
278 theta_f => grid_x%theta_F
279 zeta_f => grid_x%zeta_F
280 end select
281
282 ! iterate over all secondary modes
283 do m = 1,n_sec
284 ! set up and normal limits
285 kdl_x = sec_x_loc(m,2:3)
286 kdl_sol = sec_sol_loc(m,2:3)
287
288 ! limit to grid ranges
289 kdl_x(1) = max(kdl_x(1),grid_x%i_min)
290 kdl_x(2) = min(kdl_x(2),grid_x%i_max)
291 kdl_sol(1) = max(kdl_sol(1),grid_sol%i_min)
292 kdl_sol(2) = min(kdl_sol(2),grid_sol%i_max)
293
294 ! convert limits to local
295 kdl_x = kdl_x - grid_x%i_min + 1
296 kdl_sol = kdl_sol - grid_sol%i_min + 1
297
298 ! cycle if mode does not exist anywhere
299 if (kdl_x(1).gt.kdl_x(2) .or. kdl_sol(1).gt.kdl_sol(2)) cycle
300
301 ! get perturbation and solution normal ranges for this mode number
302 r_x_loc => grid_x%loc_r_F(kdl_x(1):kdl_x(2))
303 r_sol_loc => grid_sol%loc_r_F(kdl_sol(1):kdl_sol(2))
304
305 ! set up table indices
306 ind_x = sec_x_loc(m,4)
307 ind_sol = sec_sol_loc(m,4)
308
309 ! Set up multiplicative factors fac_0 and fac_1 which are used in
310 ! XUQ = fac_0*X + fac_1*DX with fac_0 and fac_1:
311 ! 1. fac_0 = 1 (i(nq-m) or i(n-iota m) for deriv)
312 ! fac_1 = 0
313 ! 2. fac_0 = U_0 (DU_0 for deriv)
314 ! fac_1 = U_1 (DU_1 for deriv)
315 ! 3. fac_0 = i(nq-m)/J or i(n-iota m)/J (no deriv)
316 ! fac_1 = 0
317 ! 4. fac_0 = DU_0/J - S (no deriv)
318 ! fac_1 = DU_1/J
319 ! where par_fac = nq-m or n-iota m.
320 ! These quantities are then multiplied by the exponent for each mode
321 ! number.
322 allocate(expon(xuq_dims(1),xuq_dims(2),kdl_sol(1):kdl_sol(2))) ! normal subset of grid_sol%loc_r_F
323 allocate(fac_0(xuq_dims(1),xuq_dims(2),kdl_sol(1):kdl_sol(2))) ! normal subset of grid_sol%loc_r_F
324 allocate(fac_1(xuq_dims(1),xuq_dims(2),kdl_sol(1):kdl_sol(2))) ! normal subset of grid_sol%loc_r_F
325 allocate(par_fac(kdl_sol(1):kdl_sol(2))) ! normal subset of grid_sol%loc_r_F
326
327 ! calculate parallel multiplicative factor par_fac and exponent
328 do kd = kdl_sol(1),kdl_sol(2)
329 ! set up local mode numbers
330 kd_tot = kd + grid_sol%i_min - 1
331 n_loc = mds_sol%n(kd_tot,ind_sol)
332 m_loc = mds_sol%m(kd_tot,ind_sol)
333#if ldebug
334 if (use_pol_flux_f .and. (m_loc.ne.sec_x_loc(m,1)) .or. &
335 &(.not.use_pol_flux_f .and. (n_loc.ne.sec_x_loc(m,1)))) then
336 ierr = 1
337 err_msg = 'mode numbers not consistent'
338 chckerr(err_msg)
339 end if
340#endif
341
342 if (use_pol_flux_f) then
343 par_fac(kd) = n_loc*jq(kd) - m_loc
344 else
345 par_fac(kd) = n_loc - m_loc*jq(kd)
346 end if
347 expon(:,:,kd) = n_loc*zeta_f(:,:,kd) - &
348 &m_loc*theta_f(:,:,kd)
349 end do
350
351 ! set up multiplicative factors in solution grid
352 select case (xuq_style)
353 case (1) ! calculating X
354 if (deriv_loc) then ! parallel derivative
355 do kd = kdl_sol(1),kdl_sol(2)
356 fac_0(:,:,kd) = iu*par_fac(kd)
357 end do
358 else
359 fac_0 = 1._dp
360 end if
361 fac_1 = 0._dp
362 case (2) ! calculating U
363 ! interpolate X-> sol or point
364 select case (x_grid_style)
365 case (1,3) ! equilibrium or enriched
366 allocate(u0(xuq_dims(1),xuq_dims(2),&
367 &kdl_sol(1):kdl_sol(2)))
368 allocate(u1(xuq_dims(1),xuq_dims(2),&
369 &kdl_sol(1):kdl_sol(2)))
370 if (deriv_loc) then ! parallel derivative
371 ierr = interp_v_spline(&
372 &x%DU_0(:,:,kdl_x(1):kdl_x(2),ind_x),u0,&
373 &r_x_loc,r_sol_loc,extrap)
374 chckerr('')
375 ierr = interp_v_spline(&
376 &x%DU_1(:,:,kdl_x(1):kdl_x(2),ind_x),u1,&
377 &r_x_loc,r_sol_loc,extrap)
378 chckerr('')
379 else
380 ierr = interp_v_spline(&
381 &x%U_0(:,:,kdl_x(1):kdl_x(2),ind_x),u0,&
382 &r_x_loc,r_sol_loc,extrap)
383 chckerr('')
384 ierr = interp_v_spline(&
385 &x%U_1(:,:,kdl_x(1):kdl_x(2),ind_x),u1,&
386 &r_x_loc,r_sol_loc,extrap)
387 chckerr('')
388 end if
389 case (2) ! solution
390 if (deriv_loc) then ! parallel derivative
391 u0 => x%DU_0(:,:,kdl_x(1):kdl_x(2),ind_x)
392 u1 => x%DU_1(:,:,kdl_x(1):kdl_x(2),ind_x)
393 else
394 u0 => x%U_0(:,:,kdl_x(1):kdl_x(2),ind_x)
395 u1 => x%U_1(:,:,kdl_x(1):kdl_x(2),ind_x)
396 end if
397 end select
398
399 ! assign
400 fac_0 = u0
401 fac_1 = u1
402
403 ! clean up
404 select case (x_grid_style)
405 case (1,3) ! equilibrium or enriched
406 deallocate(u0,u1)
407 case (2) ! solution
408 ! do nothing
409 end select
410 case (3) ! calculating Qn
411 if (deriv_loc) then ! parallel derivative
412 ierr = 1
413 err_msg = 'For XUQ_style = '//trim(i2str(xuq_style))//&
414 &' calculation of parallel derivative not supported'
415 chckerr('')
416 else
417 do kd = kdl_sol(1),kdl_sol(2)
418 fac_0(:,:,kd) = iu*par_fac(kd)*inv_j(:,:,kd)
419 end do
420 end if
421 fac_1 = 0._dp
422 case (4) ! calculating Qg
423 ! interpolate X-> sol or point
424 err_msg = 'For XUQ_style = '//trim(i2str(xuq_style))//&
425 &' calculation of parallel derivative not supported'
426 chckerr('')
427 select case (x_grid_style)
428 case (1,3) ! equilibrium or enriched
429 allocate(u0(xuq_dims(1),xuq_dims(2),&
430 &kdl_sol(1):kdl_sol(2)))
431 allocate(u1(xuq_dims(1),xuq_dims(2),&
432 &kdl_sol(1):kdl_sol(2)))
433 if (deriv_loc) then ! parallel derivative
434 ierr = 1
435 chckerr(err_msg)
436 else
437 ierr = interp_v_spline(&
438 &x%DU_0(:,:,kdl_x(1):kdl_x(2),ind_x),u0,&
439 &r_x_loc,r_sol_loc,extrap)
440 chckerr('')
441 ierr = interp_v_spline(&
442 &x%DU_1(:,:,kdl_x(1):kdl_x(2),ind_x),u1,&
443 &r_x_loc,r_sol_loc,extrap)
444 chckerr('')
445 end if
446 case (2) ! solution
447 if (deriv_loc) then ! parallel derivative
448 ierr = 1
449 chckerr(err_msg)
450 else
451 u0 => x%DU_0(:,:,kdl_x(1):kdl_x(2),ind_x)
452 u1 => x%DU_1(:,:,kdl_x(1):kdl_x(2),ind_x)
453 end if
454 end select
455
456 ! assign
457 fac_0 = u0*inv_j(:,:,kdl_sol(1):kdl_sol(2)) - &
458 &s(:,:,kdl_sol(1):kdl_sol(2))
459 fac_1 = u1*inv_j(:,:,kdl_sol(1):kdl_sol(2))
460
461 ! clean up
462 select case (x_grid_style)
463 case (1,3) ! equilibrium or enriched
464 deallocate(u0,u1)
465 case (2) ! solution
466 ! do nothing
467 end select
468 case default
469 err_msg = 'No XUQ style associated with '//&
470 &trim(i2str(xuq_style))
471 ierr = 1
472 chckerr(err_msg)
473 end select
474
475 ! iterate over all normal points in solution modes variables
476 do kd = kdl_sol(1),kdl_sol(2)
477 ! set up loc complex XUQ without time at this normal point
478 xuq_loc = exp(iu*expon(:,:,kd)) * &
479 &(sol%vec(ind_sol,kd,x_id)*fac_0(:,:,kd) + &
480 &dsol_vec(ind_sol,kd)*fac_1(:,:,kd))
481
482 ! iterate over time steps
483 do id = 1,n_t
484 ! add current mode
485 xuq(:,:,kd,id) = xuq(:,:,kd,id) + xuq_loc * &
486 &exp(iu*sqrt_sol_val_norm*time(id)*2*pi)
487 end do
488 end do
489
490 ! clean up
491 deallocate(fac_0,fac_1,expon,par_fac)
492 end do
493
494 ! clean up
495 select case (x_grid_style)
496 case (1,3) ! equilibrium or enriched
497 deallocate(theta_f,zeta_f)
498 case (2) ! solution
499 ! do nothing
500 end select
501 nullify(r_x_loc,r_sol_loc)
502 nullify(theta_f,zeta_f)
503 nullify(u0,u1)
504 nullify(sec_x_loc,sec_sol_loc)
505 end function calc_xuq_arr

546 !> Calculate the total version of the solution vector from the local
547 !! version.
548 !!
549 !! This is the solution vector for all of the possible mode numbers, which
550 !! can be different from the local mode numbers for X style 2 (fast):
551 !! - local: the size of the saved perturbation and solution variables,
552 !! prescribed by the user as input:
553 !! - either directly through \c n_mod_X
554 !! - or through the limits on the modes using \c min_sec_X and \c
555 !! max_sec_X.
556 !! - total: all the possible modes that can resonate in the plasma, which
557 !! can be different from the local number for \c X_style 2, as the mode
558 !! numbers depend on the normal coordinate.
559 !!
560 !! If the output variable is not allocated, it is done here.
561 !!
562 !! Optionally a derivative can be requested, depending on the normal
563 !! discretization order.
564 !!
565 !! \note Need to provide the grid in which mds is tabulated.
566 !!
567 !! \return ierr
568 integer function calc_tot_sol_vec(mds,grid_sol,sol_vec_loc,sol_vec_tot,&
569 &deriv) result(ierr)
570 use num_vars, only: x_style, norm_disc_prec_sol
571 use x_vars, only: n_mod_x
572 use x_utilities, only: get_n_mod_tot
573 use spline_utilities, only: spline
574
575 character(*), parameter :: rout_name = 'calc_tot_sol_vec'
576
577 ! input / output
578 type(modes_type), intent(in) :: mds !< general modes variables
579 type(grid_type), intent(in) :: grid_sol !< solution grid
580 complex(dp), intent(in) :: sol_vec_loc(:,:) !< solution vector for local resonating modes
581 complex(dp), intent(inout), allocatable :: sol_vec_tot(:,:) !< solution vector for all possible resonating modes
582 integer, intent(in), optional :: deriv !< order of derivative
583
584 ! local variables
585 character(len=max_str_ln) :: err_msg ! error message
586 integer :: n_mod_loc ! local number of modes that are used in the calculations
587 integer :: n_mod_tot ! total number of modes
588 integer :: ld ! counter
589 integer :: kdl(2) ! kd limits
590 integer :: ld_loc ! local ld
591 integer :: n_r ! normal size of variables
592 integer :: deriv_loc ! local derivative
593 integer :: max_deriv ! maximum derivative
594 integer :: min_range ! minimum range for derivative
595 complex(dp), allocatable :: dsol_vec_loc(:) ! local Dsol_vec_tot in solution grid
596
597 ! initialize ierr
598 ierr = 0
599
600 ! set n_r
601 n_r = size(sol_vec_loc,2)
602
603 ! set local derivative
604 deriv_loc = 0
605 if (present(deriv)) deriv_loc = deriv
606
607 ! test
608 select case (norm_disc_prec_sol)
609 case (1) ! linear
610 max_deriv = 1
611 min_range = 2
612 case (2) ! Akita hermite
613 max_deriv = 1
614 min_range = 4
615 case (3) ! spline
616 max_deriv = 3
617 min_range = 4
618 case default
619 ierr = 1
620 err_msg = 'normal discretization precision for solution &
621 &cannot be higher than 3 or lower than 0'
622 chckerr(err_msg)
623 end select
624 if (deriv_loc.lt.0 .or. deriv_loc.gt.max_deriv) then
625 ierr = 1
626 err_msg = 'deriv must be between 0 and '//trim(i2str(max_deriv))//&
627 &' for normal discretization '//trim(i2str(norm_disc_prec_sol))
628 chckerr(err_msg)
629 endif
630
631 ! operations depending on X style
632 select case (x_style)
633 case (1) ! prescribed
634 ! test allocation
635 if (allocated(sol_vec_tot)) then
636 if (size(sol_vec_tot,1).ne.size(sol_vec_loc,1) .or. &
637 &size(sol_vec_tot,2).ne.size(sol_vec_loc,2)) then
638 ierr = 1
639 err_msg = 'Local and total solution vectors need to &
640 &be of the same length'
641 chckerr(err_msg)
642 end if
643 else
644 allocate(sol_vec_tot(size(sol_vec_loc,1),n_r))
645 end if
646
647 ! copy or derivate
648 if (deriv_loc.eq.0) then ! copy
649 sol_vec_tot = sol_vec_loc
650 else ! derivate
651 do ld = 1,size(sol_vec_tot,1)
652 ierr = spline(grid_sol%r_F,sol_vec_loc(ld,:),&
653 &grid_sol%r_F,sol_vec_tot(ld,:),&
654 &ord=norm_disc_prec_sol,deriv=deriv_loc)
655 chckerr('')
656 end do
657 end if
658 case (2) ! fast
659 ! set local and total nr. of modes
660 n_mod_loc = n_mod_x
661 n_mod_tot = get_n_mod_tot(mds)
662
663 ! test allocation
664 if (allocated(sol_vec_tot)) then
665 if (size(sol_vec_tot,1).ne.n_mod_tot .or. &
666 &size(sol_vec_tot,2).ne.size(sol_vec_loc,2)) then
667 ierr = 1
668 err_msg = 'Total solution vectors needs to be of &
669 &correct length '//trim(i2str(n_mod_tot))
670 chckerr(err_msg)
671 end if
672 else
673 allocate(sol_vec_tot(n_mod_tot,n_r))
674 end if
675
676 ! track all possible modes and copy or derivate
677 sol_vec_tot = 0._dp
678 do ld = 1,size(mds%sec,1)
679 ld_loc = mds%sec(ld,1)-minval(mds%sec(:,1),1)+1
680 kdl(1) = max(mds%sec(ld,2)-grid_sol%i_min+1,1) ! normal range in solution vector
681 kdl(2) = min(mds%sec(ld,3)-grid_sol%i_min+1,n_r)
682 if (kdl(1).gt.kdl(2)) cycle ! no point of this range in sol_vec_loc
683 if (deriv_loc.eq.0) then
684 sol_vec_tot(ld_loc,kdl(1):kdl(2)) = &
685 &sol_vec_loc(mds%sec(ld,4),kdl(1):kdl(2))
686 else
687 allocate(dsol_vec_loc(kdl(2)-kdl(1)+1))
688 dsol_vec_loc = 0._dp
689
690 if (kdl(2)-kdl(1)+1.ge.min_range) then ! only calculate if enough points
691 ierr = spline(grid_sol%r_F(kdl(1):kdl(2)),&
692 &sol_vec_loc(mds%sec(ld,4),kdl(1):kdl(2)),&
693 &grid_sol%r_F(kdl(1):kdl(2)),dsol_vec_loc,&
694 &ord=norm_disc_prec_sol,deriv=deriv_loc)
695 chckerr('')
696 end if
697 sol_vec_tot(ld_loc,kdl(1):kdl(2)) = dsol_vec_loc
698
699 deallocate(dsol_vec_loc)
700 end if
701 end do
702 end select
703 end function calc_tot_sol_vec

873 !> Interpolate tensorial perturbation quantities in the third
874 !! dimension.
875 !!
876 !! The input grid should not be divided, though the output grid can be.
877 !!
878 !! The procedure considers all possible mode number combinations. For \c
879 !! X_style 2 (fast), each secondary mode only lives in a certain normal
880 !! range of the plasma. Therefore, each secondary mode pair also has a
881 !! limited normal range, given by the overlap of the ranges of the members.
882 !!
883 !! The interpolated mode number combinations have a normal range that might
884 !! slightly differ from the input ranges, in which case extrapolation can be
885 !! done if the method allows for it.
886 !!
887 !! Row \c k of \c sec of the input grid and row \c k of the output grid
888 !! belong to the same mode, because setup_modes() builds the rows of every
889 !! grid from the window on the equilibrium points.
890 !!
891 !! \note A mode pair can have output points but no input points, when the
892 !! window moves by two or more between two points of the input grid. It
893 !! is then set to zero. setup_modes() stops when the window moves by more
894 !! than \c n_mod_X between two points of a grid.
895 integer function interp_v(mds_i,grid_i,X_i,mds_o,grid_o,X_o) result(ierr)
896 use x_utilities, only: is_necessary_x, check_same_modes
897 use x_vars, only: n_mod_x
898 use num_utilities, only: c
899#if ldebug
900 use num_vars, only: rank
901 use eq_vars, only: max_flux_f
902#endif
903
904 character(*), parameter :: rout_name = 'interp_V'
905
906 ! input / output
907 type(modes_type), intent(in), target :: mds_i !< general modes variables for input
908 type(grid_type), intent(in) :: grid_i !< grid at which \c X_i is tabulated
909 type(x_2_type), intent(in), target :: x_i !< tensorial perturbation variable on input grid
910 type(modes_type), intent(in), target :: mds_o !< general modes variables for output
911 type(grid_type), intent(in) :: grid_o !< grid at which \c X_o is interpolated
912 type(x_2_type), intent(inout), target :: x_o !< interpolated tensorial perturbation variable
913
914 ! local variables
915 integer :: kd ! counter
916 integer :: k, m ! counters for mode numbers
917 integer :: c_loc_i(2) ! local input c for symmetric and asymmetric variables
918 integer :: c_loc_o(2) ! local output c for symmetric and asymmetric variables
919 integer :: n_mod_tot ! total amount of modes
920 integer :: kdl_i(2), kdl_o(2) ! limits on normal index for a mode combination
921 integer, pointer :: sec_i_loc(:,:), sec_o_loc(:,:) ! pointers to secondary mode variables
922 real(dp), pointer :: r_i_loc(:), r_o_loc(:) ! local r_i and r_o for a mode combination
923 complex(dp), pointer :: v_i(:,:,:), v_o(:,:,:) ! pointers to input and output PV_i and KV_i
924 logical :: calc_this(2) ! whether this combination needs to be calculated
925 logical :: extrap = .true. ! whether extrapolation is used
926 character(len=max_str_ln) :: err_msg ! error message
927#if ldebug
928 integer :: km_id
929 integer, allocatable :: norm_ext_i(:,:) ! normal extent for input quantity mode combinations
930 real(dp), allocatable :: r_loc_tot(:,:,:) ! r_i_loc and r_o_loc for all combinations
931 integer :: n_copy ! mode pairs with one input point (copied)
932 !complex(dp), allocatable :: V_plot(:,:) ! for debug plotting of interpolated V
933#endif
934
935 ! initialize ierr
936 ierr = 0
937
938 ! test
939 if (grid_i%n(2).ne.grid_o%n(2)) then
940 ierr = 1
941 err_msg = 'input and output grid are not compatible in the &
942 &geodesic coordinate: '//trim(i2str(grid_i%n(2)))//' vs. '//&
943 &trim(i2str(grid_o%n(2)))
944 chckerr(err_msg)
945 end if
946
947 ! initialize to zero
948 x_o%PV_0 = 0._dp
949 x_o%PV_1 = 0._dp
950 x_o%PV_2 = 0._dp
951 x_o%KV_0 = 0._dp
952 x_o%KV_1 = 0._dp
953 x_o%KV_2 = 0._dp
954
955 ! the input and output grid have the same rows of modes
956 ierr = check_same_modes(mds_i,mds_o)
957 chckerr('')
958 sec_i_loc => mds_i%sec
959 sec_o_loc => mds_o%sec
960
961 ! select all mode number combinations of interpolated grid
962 n_mod_tot = size(sec_o_loc,1)
963#if ldebug
964 allocate(r_loc_tot(2,n_mod_tot**2,3))
965 allocate(norm_ext_i(n_mod_tot,n_mod_tot))
966 km_id = 0
967 norm_ext_i = 0
968 n_copy = 0
969#endif
970 do m = 1,n_mod_tot
971 do k = 1,n_mod_tot
972 ! set input and output grid normal limits for mode pair (k,m)
973 kdl_i(1) = max(sec_i_loc(k,2),sec_i_loc(m,2))
974 kdl_i(2) = min(sec_i_loc(k,3),sec_i_loc(m,3))
975 kdl_o(1) = max(sec_o_loc(k,2),sec_o_loc(m,2))
976 kdl_o(2) = min(sec_o_loc(k,3),sec_o_loc(m,3))
977
978 ! limit to grid ranges
979 kdl_i(1) = max(kdl_i(1),grid_i%i_min)
980 kdl_i(2) = min(kdl_i(2),grid_i%i_max)
981 kdl_o(1) = max(kdl_o(1),grid_o%i_min)
982 kdl_o(2) = min(kdl_o(2),grid_o%i_max)
983
984 ! limit output range to input range if not extrapolating
985 if (.not.extrap) then
986 do kd = kdl_o(1),kdl_o(2)
987 if (grid_o%r_F(kd).ge.grid_i%r_F(kdl_i(1))) exit
988 end do
989 kdl_o(1) = kd
990
991 do kd = kdl_o(2),kdl_o(1),-1
992 if (grid_o%r_F(kd).le.grid_i%r_F(kdl_i(2))) exit
993 end do
994 kdl_o(2) = kd
995 end if
996
997 ! cycle if mode combination does not exist anywhere
998 if (kdl_i(1).gt.kdl_i(2) .or. kdl_o(1).gt.kdl_o(2)) cycle
999
1000 ! convert limits to local
1001 kdl_i = kdl_i - grid_i%i_min + 1
1002 kdl_o = kdl_o - grid_o%i_min + 1
1003
1004 ! get input ranges for this mode number
1005 r_i_loc => grid_i%loc_r_F(kdl_i(1):kdl_i(2))
1006 r_o_loc => grid_o%loc_r_F(kdl_o(1):kdl_o(2))
1007
1008 ! check whether mode combination needs to be calculated
1009 calc_this(1) = is_necessary_x(.true.,&
1010 &[sec_o_loc(k,1),sec_o_loc(m,1)])
1011 calc_this(2) = is_necessary_x(.false.,&
1012 &[sec_o_loc(k,1),sec_o_loc(m,1)])
1013
1014#if ldebug
1015 km_id = km_id + 1
1016 r_loc_tot(:,km_id,1) = [r_i_loc(1),r_i_loc(size(r_i_loc))]
1017 r_loc_tot(:,km_id,2) = [r_o_loc(1),r_o_loc(size(r_o_loc))]
1018 r_loc_tot(:,km_id,3) = km_id
1019 norm_ext_i(k,m) = size(r_i_loc)
1020 if (debug_interp_v) write(*,*) 'k, m', k, m, 'of', n_mod_tot, &
1021 &'with calc_this = ', calc_this
1022#endif
1023
1024 ! set up c_loc for input and output
1025 c_loc_i(1) = c([sec_i_loc(k,4),sec_i_loc(m,4)],.true.,n_mod_x)
1026 c_loc_i(2) = c([sec_i_loc(k,4),sec_i_loc(m,4)],.false.,n_mod_x)
1027 c_loc_o(1) = c([sec_o_loc(k,4),sec_o_loc(m,4)],.true.,n_mod_x)
1028 c_loc_o(2) = c([sec_o_loc(k,4),sec_o_loc(m,4)],.false.,n_mod_x)
1029
1030 if (calc_this(1)) then
1031 v_i => x_i%PV_0(:,:,kdl_i(1):kdl_i(2),c_loc_i(1))
1032 v_o => x_o%PV_0(:,:,kdl_o(1):kdl_o(2),c_loc_o(1))
1033 ierr = interp_v_spline(v_i,v_o,r_i_loc,r_o_loc,extrap)
1034 chckerr('')
1035
