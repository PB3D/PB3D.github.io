44 !> Calculate G_ij and H_ij on an interval for field line aligned
45 !! configurations.
46 !!
47 !! The source variables are given on a stencil of four consecutive points
48 !! of the field line, <tt>[1:4,dim]</tt> with \c dim the Cartesian dimension;
49 !! the interval is the one between stencil points \c i_int and \c i_int+1.
50 !! Each endpoint receives the kernel integrated over its half of the
51 !! interval, normalized to the half-cell weight \f$\Delta p \Delta a /
52 !! 2\f$ that calc_GH_1 applies, so that a constant kernel returns its
53 !! value.
54 !!
55 !! Far from the influence point, more than \c n_near interval lengths away,
56 !! the kernel value at the endpoint is used, which amounts to the
57 !! trapezoidal rule along the field line. Closer, the kernel is
58 !! integrated over the half-interval with 8-point Gauss-Legendre on
59 !! the cubic interpolant of position and normal through the stencil.
60 !! This resolves the variation of the kernel along a neighbouring field
61 !! line: where the field lines run almost toroidally, as at the bean tips
62 !! of W7-X, they are packed far closer (0.007 \f$R_0\f$) than the points
63 !! along them (0.06 \f$R_0\f$), the kernel seen along the neighbouring line
64 !! is a peak much narrower than the point spacing, and sampling it at the
65 !! points gives errors of order one whose sign depends on how the
66 !! points of neighbouring lines happen to line up. The stencil and the
67 !! Gauss points lie on one field line, so this rule changes the integral
68 !! along the line only; across the field lines calc_GH_1 keeps the sum
69 !! with the weight \f$\Delta a\f$ per field line.
70 !!
71 !! For subintegrals adjacent to the singularity (i.e. with the influence
72 !! point as one of their endpoints), the contribution of the singular
73 !! endpoint is replaced by the analytical integral of the kernel over its
74 !! grid half-cell. In field-aligned coordinates, with \f$\delta p\f$ along
75 !! the field line (metric \f$g_{\theta\theta}\f$ = <tt>h_fac(3)</tt>) and
76 !! \f$\delta a\f$ the field line label (metric \f$g_{\alpha\alpha}\f$ =
77 !! <tt>h_fac(1)</tt>, coupling \f$g_{\alpha\theta}\f$ = <tt>h_fac(2)</tt>),
78 !! the squared distance near the singular point is the quadratic form
79 !! \f$ g_{\theta\theta} \delta p^2 + 2 g_{\alpha\theta} \delta p \delta a
80 !! + g_{\alpha\alpha} \delta a^2 \f$,
81 !! and with the metric-normalized half-cell sizes \f$U = \sqrt{g_{\theta
82 !! \theta}} \Delta p/2\f$, \f$V = \sqrt{g_{\alpha\alpha}} \Delta a/2\f$ and
83 !! nonorthogonality \f$s = g_{\alpha\theta} / \sqrt{g_{\alpha\alpha}
84 !! g_{\theta\theta}}\f$, the exact primitive is
85 !! \f[ \int_0^U \int_{-V}^{V} \frac{\text{d}u \text{d}v}
86 !! {\sqrt{u^2+2suv+v^2}} =
87 !! U \ln\left(\frac{V+sU+D_+}{-V+sU+D_-}\right) +
88 !! V \ln\left(\frac{U+sV+D_+}{-U+sV+D_-}\right), \quad
89 !! D_\pm = \sqrt{U^2 \pm 2sUV + V^2}. \f]
90 !! In the same units as the regular kernel values, the singular endpoint
91 !! contribution is then
92 !! \f$-2 F / (\Delta p \Delta a \sqrt{g_{\alpha\alpha} g_{\theta\theta}})
93 !! \f$.
94 !!
95 !! (Verified against brute-force quadrature in the full-stack tests.)
96 !!
97 !! \note This routine does not calculate the contribution \f$2\beta\f$ nor
98 !! the cell integral of the H kernel at the singular point: vac_ops.calc_gh()
99 !! sets the diagonals afterwards. By default both come from the identities
100 !! of vac_ops.diag_indirect(), which replace the half-cell value of G computed
101 !! here; with \c vac_diag_style 2 the diagonal of H comes from
102 !! calc_diag_direct() and G keeps the half-cell value.
103 !!
104 !! \note With \c use_near false the near-field rule is not used. For
105 !! points off the boundary (vac_ops.eval_pot_ext()) it pays only within
106 !! about an interval length of the boundary: farther out the trapezoidal
107 !! rule along the lines and across them is more accurate, because the
108 !! errors of the trapezoidal rule at the ends of the lines cancel between
109 !! the lines, whose ends interleave, while the edge of the near-field zone
110 !! leaves an error of second order in the step along the lines (on the
111 !! circular VMEC twin at 64 lines of 100 points: 1e-3 of the potential at
112 !! half a minor radius from the boundary, against 1e-7 with the
113 !! trapezoidal rule alone).
114 subroutine calc_gh_int_1(G,H,x_s,x_in,norm_s,h_fac_in,step_size,tol,i_int,&
115 &use_near)
116 ! input / output
117 real(dp), intent(inout) :: g(2) !< G
118 real(dp), intent(inout) :: h(2) !< H
119 real(dp), intent(in) :: x_s(4,3) !< position vector at the four stencil points
120 real(dp), intent(in) :: x_in(3) !< position vector at which to calculate influence
121 real(dp), intent(in) :: norm_s(4,3) !< normal vector at the four stencil points
122 real(dp), intent(in) :: h_fac_in(3) !< metric factors g_aa, g_at, g_tt at the influence point
123 real(dp), intent(in) :: step_size(2) !< step sizes in parallel direction and alpha
124 real(dp), intent(in) :: tol !< tolerance on the squared distance between an endpoint and the influence point
125 integer, intent(in) :: i_int !< stencil index of the left point of the interval (1, 2 or 3)
126 logical, intent(in), optional :: use_near !< whether the near-field rule may be used (default true)
127
128 ! local variables
129 integer, parameter :: n_gl = 8 ! Gauss-Legendre points per half-interval
130 real(dp), parameter :: n_near = 4._dp ! near-field range in interval lengths
131 ! (a fixed count; issue #9 re-evaluates it)
132 real(dp), parameter :: gl_x(n_gl) = [&
133 &-0.9602898564975363_dp, -0.7966664774136267_dp, &
134 &-0.5255324099163290_dp, -0.1834346424956498_dp, &
135 & 0.1834346424956498_dp, 0.5255324099163290_dp, &
136 & 0.7966664774136267_dp, 0.9602898564975363_dp] ! Gauss-Legendre abscissas on [-1,1]
137 real(dp), parameter :: gl_w(n_gl) = [&
138 &0.1012285362903763_dp, 0.2223810344533745_dp, &
139 &0.3137066458778873_dp, 0.3626837833783620_dp, &
140 &0.3626837833783620_dp, 0.3137066458778873_dp, &
141 &0.2223810344533745_dp, 0.1012285362903763_dp] ! Gauss-Legendre weights on [-1,1]
142 integer :: kd, jd ! counters
143 real(dp) :: r2(2) ! distance between source and influence points
144 real(dp) :: s_no ! nonorthogonality g_at/sqrt(g_aa g_tt)
145 real(dp) :: u_hc, v_hc ! metric-normalized half-cell sizes
146 real(dp) :: d_pm(2) ! D_+ and D_-
147 real(dp) :: f_int ! primitive of the half-cell integral
148 real(dp) :: s, l(4) ! stencil parameter and cubic Lagrange weights
149 real(dp) :: d_q(3), n_q(3), r_q ! relative position, normal and distance at a quadrature point
150 logical :: near ! interval within n_near lengths of the influence point
151
152 ! initialize
153 g = 0._dp
154 h = 0._dp
155
156 ! calculate distance between source points
157 do kd = 1,2
158 r2(kd) = sum((x_s(i_int+kd-1,:)-x_in)**2)
159 end do
160 near = minval(r2).lt.n_near**2*sum((x_s(i_int+1,:)-x_s(i_int,:))**2)
161 if (present(use_near)) near = near .and. use_near
162
163 do kd = 1,2
164 if (r2(kd).le.tol) then ! singular endpoint: analytical half-cell integral
165 s_no = h_fac_in(2)/sqrt(h_fac_in(1)*h_fac_in(3))
166 u_hc = sqrt(h_fac_in(3))*step_size(1)/2._dp
167 v_hc = sqrt(h_fac_in(1))*step_size(2)/2._dp
168 d_pm(1) = sqrt(u_hc**2+2*s_no*u_hc*v_hc+v_hc**2)
169 d_pm(2) = sqrt(u_hc**2-2*s_no*u_hc*v_hc+v_hc**2)
170 f_int = u_hc*log(abs((v_hc+s_no*u_hc+d_pm(1))/&
171 &(-v_hc+s_no*u_hc+d_pm(2)))) + &
172 &v_hc*log(abs((u_hc+s_no*v_hc+d_pm(1))/&
173 &(-u_hc+s_no*v_hc+d_pm(2))))
174 g(kd) = -2._dp*f_int/(step_size(1)*step_size(2)*&
175 &sqrt(h_fac_in(1)*h_fac_in(3)))
176 h(kd) = 0._dp ! the diagonal of H is set by calc_GH
177 else if (near) then ! near endpoint: kernel integrated over its half-interval
178 do jd = 1,n_gl
179 s = i_int-1 + 0.25_dp*(2*kd-1+gl_x(jd)) ! stencil parameter, nodes at 0, 1, 2, 3
180 l = [-(s-1)*(s-2)*(s-3)/6._dp, s*(s-2)*(s-3)/2._dp, &
181 &-s*(s-1)*(s-3)/2._dp, s*(s-1)*(s-2)/6._dp]
182 d_q = matmul(l,x_s) - x_in
183 n_q = matmul(l,norm_s)
184 r_q = sqrt(sum(d_q**2))
185 g(kd) = g(kd) - 0.5_dp*gl_w(jd)/r_q
186 h(kd) = h(kd) + 0.5_dp*gl_w(jd)*sum(n_q*d_q)/r_q**3
187 end do
188 else ! far endpoint: kernel value
189 g(kd) = -1._dp/sqrt(r2(kd))
190 ! dipole kernel norm.(x_s-x_in)/|x_s-x_in|^3, i.e. the
191 ! directional derivative of G along the source normal
192 h(kd) = sum(norm_s(i_int+kd-1,:)*(x_s(i_int+kd-1,:)-x_in))*&
193 &(-g(kd))**3
194 end if
195 end do
196 end subroutine calc_gh_int_1

198 !> Calculate G_ij and H_ij on an interval for axisymmetric configurations.
199 !!
200 !! The indices for the source variables are <tt>[left:right,dim]</tt> where
201 !! \c dim is the Cartesian dimension. The same holds for \c ql.
202 !!
203 !! For subintegrals close to the singularity (i.e. where the tolerance is no
204 !! being met), the routine approximates the toroidal functions with the
205 !! analytical approximation.
206 !! If one of the boundaries of the subintegral is far enough from the
207 !! singularity, there is an additional contribution due to the difference
208 !! between the actual toroidal function and the approximation. This
209 !! contribution is zero if both boundaries are close, because in this case
210 !! it is assumed that the entire integral is given by the analytical
211 !! approximation.
212 !!
213 !! \note The kernels are those of eqs. (130)-(131) of the 2019 manuscript
214 !! with the sign of H reversed (the leading minus signs of the three H
215 !! formulas): the normal derivative is taken along the stored normal
216 !! \f$-\mathcal{J}\nabla\psi\f$, out of the vacuum, for both vacuum styles
217 !! (erratum E1 of that manuscript). This routine does not
218 !! calculate the corner term: calc_GH_2 adds \f$-2\beta\f$ to the diagonal
219 !! of H, with \f$\beta\f$ the vacuum's internal angle
220 !! on the vacuum side.
221 subroutine calc_gh_int_2(G,H,t_s,t_in,x_s,x_in,norm_s,norm_in,Aij,ql,tol,&
222 &b_coeff,n_tor)
223 ! input / output
224 real(dp), intent(inout) :: g(2) !< G
225 real(dp), intent(inout) :: h(2) !< H
226 real(dp), intent(in) :: t_s(2) !< pol. angle at left and right limit of source interval
227 real(dp), intent(in) :: t_in !< pol. angle at which to calculate influence
228 real(dp), intent(in) :: x_s(2,2) !< position vector at left and right limit of source interval
229 real(dp), intent(in) :: x_in(2) !< position vector at which to calculate influence
230 real(dp), intent(in) :: norm_s(2,2) !< normal vector at left and right limit of source interval
231 real(dp), intent(in) :: norm_in(2) !< normal vector at which to calculate influence
232 real(dp), intent(in) :: aij(2) !< Aij helper variable for H
233 real(dp), intent(in) :: ql(2,2) !< ql for n-1 and n at left and right limit of source interval
234 real(dp), intent(in) :: tol !< tolerance on rho2
235 real(dp), intent(in) :: b_coeff(2) !< \f$b = \sum_{k=1}^n \frac{2}{2k-1}\f$ for n and n-1
236 integer, intent(in) :: n_tor !< toroidal mode number
237
238 ! local variables
239 integer :: kd, jd ! counters
240 integer :: kd_bound ! counter for boundary
241 real(dp) :: a_coeff ! coefficient a = |norm|/8R^2 at influence point
242 real(dp) :: t_in_loc ! local t_in, possibly shifted by 2pi
243 real(dp) :: delta_t(2) ! t_in - t_s
244 real(dp) :: rho2(2) ! square of distance in projected poloidal plane
245 real(dp) :: dlogd(2) ! delta log delta
246
247 ! initialize
248 g = 0._dp
249 h = 0._dp
250 do kd = 1,2
251 rho2(kd) = sum((x_s(kd,:)-x_in)**2)
252 end do
253 if (pi .lt. sum(t_s)/2 - t_in) then ! t_s closer to t_in + 2pi
254 t_in_loc = t_in + 2*pi
255 else if (-pi .gt. sum(t_s)/2 - t_in) then ! t_s closer to t_in - 2pi
256 t_in_loc = t_in - 2*pi
257 else ! t_s closer to t_in
258 t_in_loc = t_in
259 end if
260
261 ! check for (near-)singular point
262 if (minval(rho2).le.tol) then
263 ! There is at least one singular point: use analytical approach
264 a_coeff = sqrt(sum(norm_in**2))/(8._dp*x_in(1)**2)
265 do kd = 1,2
266 if (kd.eq.1) then
267 delta_t(1) = t_s(1) - t_in_loc
268 delta_t(2) = t_s(2) - t_in_loc
269 else
270 delta_t(1) = -(t_s(2) - t_in_loc)
271 delta_t(2) = -(t_s(1) - t_in_loc)
272 end if
273 do jd = 1,2
274 if (abs(delta_t(jd)).le.0._dp) then
275 dlogd(jd) = 0._dp
276 else
277 dlogd(jd) = log(a_coeff*abs(delta_t(jd)))
278 end if
279 end do
280 g(kd) = 1._dp/sqrt(x_in(1)*x_s(kd,1)) * (&
281 &b_coeff(2)*(delta_t(2)-delta_t(1)) + &
282 &0.5_dp*delta_t(1) - 1.5_dp*delta_t(2) + &
283 &1._dp/(delta_t(2)-delta_t(1)) * &
284 &(delta_t(1)*(delta_t(1)-2*delta_t(2))*dlogd(1) + &
285 &delta_t(2)**2 * dlogd(2)) &
286 &)
287 h(kd) = -0.5*norm_s(kd,1)/x_s(kd,1) * g(kd) - &
288 &(delta_t(2)-delta_t(1)) * (n_tor-0.5_dp)**(-1) * &
289 &aij(kd)/sqrt(x_in(1)*x_s(kd,1))
290 end do
291
292 ! Check if we are at the boundary of the analytical approach. In
293 ! this case there is an extra component due to the difference
294 ! between the toroidal function and the analytical approximation.
295 ! This will be implemented through a trapezoidal contribution of
296 ! which one side is zero.
297 if (maxval(rho2).gt.tol) then
298 ! fill G's and H's
299 if (rho2(2).gt.rho2(1)) then ! right boundary
300 kd_bound = 2
301 else ! left boundary
302 kd_bound = 1
303 end if
304 g(kd_bound) = g(kd_bound) - 0.5_dp*(t_s(2)-t_s(1))*2._dp/&
305 &sqrt(x_in(1)*x_s(kd_bound,1)) * (ql(kd_bound,2)+&
306 &log(a_coeff*abs(t_s(kd_bound)-t_in_loc))+b_coeff(2))
307 h(kd_bound) = h(kd_bound) - 0.5_dp*(t_s(2)-t_s(1))*2._dp/&
308 &sqrt(x_in(1)*x_s(kd_bound,1)) * &
309 &( (-0.5_dp*norm_s(kd_bound,1)/x_s(kd_bound,1) - &
310 &(1._dp+rho2(kd_bound)/(2*x_in(1)*x_s(kd_bound,1)))*&
311 &aij(kd_bound)) * (ql(kd_bound,2)+&
312 &log(a_coeff*abs(t_s(kd_bound)-t_in_loc))+b_coeff(2)) + &
313 &aij(kd_bound) * (ql(kd_bound,1)+&
314 &log(a_coeff*abs(t_s(kd_bound)-t_in_loc))+b_coeff(1)))
315 end if
316 else
317 do kd = 1,2
318 ! fill G's and H's
319 g(kd) = - (t_s(2)-t_s(1))/sqrt(x_in(1)*x_s(kd,1)) * &
320 &ql(kd,2)
321 h(kd) = -(t_s(2)-t_s(1))/sqrt(x_in(1)*x_s(kd,1)) * &
322 &( (-0.5_dp*norm_s(kd,1)/x_s(kd,1) - &
323 &(1._dp+rho2(kd)/(2*x_in(1)*x_s(kd,1)))*&
324 &aij(kd))*ql(kd,2) + aij(kd)*ql(kd,1) )
325 end do
326 end if
327 end subroutine calc_gh_int_2

329 !> Gathers a distributed vector on a single process.
330 !!
331 !! For the distributed vector, the limits of the different subrows need to
332 !! be provided, as discussed in init_vac().
333 !!
334 !! By default, the result lands on the master process, but this can be
335 !! changed using \c proc. If it is negative, all processes receive the
336 !! result. In any case, \c vec_loc will be utilized.
337 !!
338 !! \note \c vec_loc needs to be allocated to the total dimensions, even when
339 !! the results is not received by the process.
340 integer function vec_dis2loc(ctxt,vec_dis,lims,vec_loc,proc) result(ierr)
341 use num_vars, only: n_procs
342 use vac_vars, only: in_context
343
344 ! input / output
345 integer, intent(in) :: ctxt !< context for vector
346 real(dp), intent(in) :: vec_dis(:) !< distributed vector
347 integer, intent(in) :: lims(:,:) !< limits for different subrows
348 real(dp), intent(inout) :: vec_loc(:) !< local vector
349 integer, intent(in), optional :: proc !< which process receives result
350
351 ! local variables
352 integer :: id ! index of subrow
353 integer :: limsl(2) ! local limits
354 integer :: proc_loc ! local proc
355 integer :: proc_dest(2) ! destination process index
356
357 ! initialize ierr
358 ierr = 0
359
360 ! select only processes that are in the context
361 if (.not.in_context(ctxt)) return
362
363 ! set up destination process index
364 proc_loc = 0
365 if (present(proc)) proc_loc = proc
366 if (proc_loc.gt.0 .and. proc_loc.lt.n_procs) then
367 call blacs_pcoord(ctxt,proc_loc,proc_dest(1),proc_dest(2))
368 else
369 proc_dest = [-1,-1]
370 end if
371
372 ! set up total copy of vector
373 vec_loc = 0._dp
374 do id = 1,size(lims,2)
375 ! set local limits
376 if (size(vec_dis) .gt. 0) then
377 limsl = sum(lims(2,1:id-1)-lims(1,1:id-1)+1) + &
378 &lims(:,id)-lims(1,id)+1
379 vec_loc(lims(1,id):lims(2,id)) = vec_dis(limsl(1):limsl(2))
380 end if
381 end do
382
383 ! add them together
384 call dgsum2d(ctxt,'all',' ',size(vec_loc),1,vec_loc,size(vec_loc),&
385 &proc_dest(1),proc_dest(2))
386 end function vec_dis2loc

398 real(dp), intent(in) :: mat_dis(:,:) !< distributed matrix
399 integer, intent(in) :: lims_r(:,:) !< limits for different subrows
400 integer, intent(in) :: lims_c(:,:) !< limits for different subcolumns
401 real(dp), intent(inout) :: mat_loc(:,:) !< local matrix
402 integer, intent(in), optional :: proc !< which process receives result
403
404 ! local variables
405 integer :: i_rd, i_cd ! index of subrow and subcolumn
406 integer :: limsl_r(2) ! local row limits
407 integer :: limsl_c(2) ! local column limits
408 integer :: proc_loc ! local proc
409 integer :: proc_dest(2) ! destination process index

411 ! initialize ierr

413
414 ! select only processes that are in the context
415 if (.not.in_context(ctxt)) return
416
417 ! set up destination process index
418 proc_loc = 0
419 if (present(proc)) proc_loc = proc
420 if (proc_loc.gt.0 .and. proc_loc.lt.n_procs) then

423 proc_dest = [-1,-1]
424 end if

426 ! set up total copy of matrix
427 mat_loc = 0._dp
428 do i_rd = 1,size(lims_r,2)

726 !> \public Solid angle at a node of the mesh, on the side of \c n_hat.
727 !!
728 !! The surface of the mesh at a node is the cone of four flat facets, one
729 !! between each pair of edges that are consecutive in azimuth about \c
730 !! n_hat. The solid angle of the region on the \c n_hat side of all four
731 !! facets is eq. (45) of the 2019 manuscript on the vacuum energy (exact;
732 !! not its small-tilt form (46), which fails on skewed meshes, erratum E6):
733 !! with the facet normals \f$\mathbf{a}_i\f$ tilted by \f$\beta_i\f$ from
734 !! \c n_hat towards the azimuth \f$\bar\varphi_i\f$, and \f$\varphi_i\f$ the
735 !! azimuths of the edges,
736 !! \f[ \Omega = 2\pi + \sum_{i=1}^4 \left[\sin^{-1}\left(\sin\beta_i
737 !! \sin(\varphi-\bar\varphi_i)\right)\right]_{\varphi_i}^{\varphi_{i+1}} . \f]
738 !! The edges may be given in any order. With \c n_hat the unit normal into
739 !! the plasma, the result is the plasma's solid angle \f$\Omega_p\f$ and
740 !! the vacuum's is \f$4\pi - \Omega_p\f$.
741 pure subroutine solid_angle_node(u,n_hat,omega)
742 ! input / output
743 real(dp), intent(in) :: u(4,3) !< the four edge vectors from the node to its neighbours
744 real(dp), intent(in) :: n_hat(3) !< unit normal of the side whose solid angle is returned
745 real(dp), intent(out) :: omega !< solid angle on that side
746
747 ! local variables
748 integer :: id, jd, kd ! counters
749 integer :: ord(4) ! edges sorted by azimuth
750 integer :: i1, i2 ! edges of a facet
751 real(dp) :: z(3), x(3), y(3) ! local frame
752 real(dp) :: uh(4,3) ! unit edges
753 real(dp) :: ph(4) ! azimuths of the edges
754 real(dp) :: a(3) ! facet normal
755 real(dp) :: c, beta, phb ! cos of the tilt, tilt and azimuth of the facet normal
756 real(dp) :: p1, p2 ! azimuths of the sector
757
758 ! local frame: z along n_hat, x along the tangential part of the first edge
759 z = n_hat/norm2(n_hat)
760 x = u(1,:) - sum(u(1,:)*z)*z
761 x = x/norm2(x)
762 y = cross3(z,x)
763
764 ! unit edges and their azimuths
765 do id = 1,4
766 uh(id,:) = u(id,:)/norm2(u(id,:))
767 ph(id) = atan2(sum(uh(id,:)*y),sum(uh(id,:)*x))
768 end do
769
770 ! sort the edges by azimuth (insertion sort)
771 ord = [1,2,3,4]
772 do id = 2,4
773 kd = ord(id)
774 jd = id-1
775 do while (jd.ge.1)
776 if (ph(ord(jd)).le.ph(kd)) exit
777 ord(jd+1) = ord(jd)
778 jd = jd-1
779 end do
780 ord(jd+1) = kd
781 end do
782
783 ! sum over the four sectors
784 omega = 2*pi
785 do id = 1,4
786 i1 = ord(id)
787 i2 = ord(mod(id,4)+1)
788 a = cross3(uh(i1,:),uh(i2,:))
789 a = a/norm2(a)
790 c = sum(a*z)
791 if (c.lt.0._dp) then ! normal of the facet on the n_hat side
792 a = -a
793 c = -c
794 end if
795 beta = acos(min(c,1._dp))
796 phb = atan2(sum(a*y),sum(a*x))
797 p1 = ph(i1)
798 p2 = ph(i2)
799 if (p2.le.p1) p2 = p2 + 2*pi ! the sector that closes the circle
800 omega = omega + asin(sin(beta)*sin(p2-phb)) - &
801 &asin(sin(beta)*sin(p1-phb))
802 end do
803 end subroutine solid_angle_node

805 !> \public Integral of the H kernel over the cell of a node of the
806 !! field-line mesh.
807 !!
808 !! The kernel \f$\mathbf{n}(\mathbf{r})\cdot(\mathbf{r}-\mathbf{r}_0)/
809 !! |\mathbf{r}-\mathbf{r}_0|^3\f$ (with \f$\mathbf{n}\f$ the stored normal
810 !! \f$-\mathcal{J}\nabla\psi\f$ at the source) is bounded on the surface
811 !! near its own node, because the surface leaves its tangent plane at
812 !! second order: to leading order \f$\mathbf{n}(\mathbf{r})\cdot
813 !! (\mathbf{r}-\mathbf{r}_0) = -\tfrac12\, \mathbf{n}_0\cdot\partial_j
814 !! \mathbf{e}_k\, \Delta u^j \Delta u^k\f$ (\f$j,k\f$ the parallel and
815 !! field-line coordinates). Over the cell of the node, \f$|\theta-\theta_0|
816 !! \le \Delta\theta/2\f$, \f$|\alpha-\alpha_0| \le \Delta\alpha/2\f$,
817 !! in the local polar coordinates of eqs. (49)-(72) of the 2019 manuscript
818 !! (with \f$A = g_{\alpha\theta}/|\mathbf{n}|\f$, \f$B = g_{\alpha\alpha}/
819 !! |\mathbf{n}|\f$ and \f$\rho(\beta)\f$ the radius of the cell), the
820 !! integral is
821 !! \f[ I = -\frac{\sqrt{g_{\alpha\alpha}}}{2|\mathbf{n}|^2}
822 !! \int_{-\pi/2}^{\pi/2} q(\beta)\,\rho(\beta)\,\mathrm{d}\beta, \quad
823 !! q = \mathbf{n}_0\cdot\partial_j\mathbf{e}_k\, t^j t^k, \quad
824 !! t = \left(\cos\beta, \frac{\sin\beta - A\cos\beta}{B}\right), \f]
825 !! (with \f$A\f$ and \f$B\f$ taken positive: the mirror cell, the frame of the
826 !! opposite orientation, is an isometric image and gives the same integral),
827 !! which is the corrected eq. (69) of the manuscript (erratum E22: the
828 !! printed one has a wrong factor and only the umbilic part of the second
829 !! fundamental form). The half-cell of the manuscript equals the full cell
830 !! of the node for this kernel. Verified against direct integration on
831 !! sphere and torus nodes (first-order convergence), and by the full-stack
832 !! test \c H_sing_node.
833 pure subroutine h_sing_node(norm0,g,dd,dpar,dalpha,I_sing)
834 ! input / output
835 real(dp), intent(in) :: norm0(3) !< stored normal at the node
836 real(dp), intent(in) :: g(3) !< metric factors g_aa, g_at, g_tt at the node
837 real(dp), intent(in) :: dd(3,3) !< second derivatives of the position: d_t e_t, d_t e_a, d_a e_a (rows)
838 real(dp), intent(in) :: dpar !< parallel step
839 real(dp), intent(in) :: dalpha !< step in the field-line label
840 real(dp), intent(out) :: i_sing !< integral of the kernel over the cell
841
842 ! local variables
843 integer :: ip, jd ! counters
844 real(dp) :: nn ! |norm0|
845 real(dp) :: a, b ! coefficients of the local frame
846 real(dp) :: q_tt, q_ta, q_aa ! second fundamental form along the stored normal
847 real(dp) :: lims(4) ! limits of the three pieces of the cell boundary
848 real(dp) :: p_lo, p_hi ! limits of a piece
849 real(dp) :: beta, c, s ! polar angle, cos, sin-like coordinate
850 real(dp) :: q, rho ! quadratic form and radius
851 real(dp) :: tot ! integral
852
853 nn = norm2(norm0)
854 a = g(2)/nn
855 b = g(1)/nn
856 q_tt = sum(norm0*dd(1,:))
857 q_ta = sum(norm0*dd(2,:))
858 q_aa = sum(norm0*dd(3,:))
859 lims = [-pi/2, atan(-b*dalpha/dpar+a), atan(b*dalpha/dpar+a), pi/2]
860 tot = 0._dp
861 do ip = 1,3
862 p_lo = lims(ip)
863 p_hi = lims(ip+1)
864 do jd = 1,n_gl16
865 beta = 0.5_dp*(p_lo+p_hi) + 0.5_dp*(p_hi-p_lo)*gl16_x(jd)
866 c = cos(beta)
867 s = (sin(beta)-a*c)/b ! field-line coordinate of the direction
868 q = q_tt*c**2 + 2*q_ta*c*s + q_aa*s**2
869 select case (ip)
870 case (1) ! -pi/2 .. beta_1: lower edge alpha = alpha_0 - dalpha
871 rho = b*dalpha/(a*c-sin(beta))
872 case (2) ! beta_1 .. beta_2: far edge theta = theta_0 + dpar
873 rho = dpar/c
874 case default ! beta_2 .. pi/2: upper edge alpha = alpha_0 + dalpha
875 rho = b*dalpha/(sin(beta)-a*c)
876 end select
877 tot = tot + 0.5_dp*(p_hi-p_lo)*gl16_w(jd)*q*rho
878 end do
879 end do
880 i_sing = -sqrt(g(1))/(2*nn**2)*tot
881 end subroutine h_sing_node

883 !> \public Direct diagonals of H for the field-line vacuum.
884 !!
885 !! Two variants, both with the cell integral of the kernel (H_sing_node):
886 !! - smooth: \f$-2\pi\f$ for the corner term, the solid angle of the

899 integer function calc_diag_direct(vac,dpar,dalpha,diag_direct,diag_facet,&

901
902 character(*), parameter :: rout_name = 'calc_diag_direct'

905 type(vac_type), intent(in) :: vac !< vacuum variables
906 real(dp), intent(in) :: dpar !< parallel step

1010 !> \public Fits diagonal entries of H and G to Green's identity for
1011 !! potentials whose boundary values and normal data are known.
1012 !!
1013 !! For a potential \f$\phi_k\f$ with normal data \f$\phi'_k\f$, row
1014 !! \f$i\f$ of the identity is
1015 !! \f[ H_{ii}\phi_{k,i} - G_{ii}\phi'_{k,i} = r_{k,i} \equiv c_k\phi_{k,i}
1016 !! - \sum_{j\ne i} H_{ij}\phi_{k,j} + \sum_{j\ne i} G_{ij}\phi'_{k,j} , \f]
1017 !! where the free term \f$c_k\f$ is \f$-4\pi\f$ for a potential that is
1018 !! harmonic inside the boundary and \f$0\f$ for one that is harmonic
1019 !! outside it and decays. The off-diagonal sums are the products with the
1020 !! full matrices minus their current diagonals, so the matrices are not
1021 !! changed.
1022 !!
1023 !! Either one entry is fitted, \c fit_H or \c fit_G, in the least-squares
1024 !! sense over any number of potentials; the other keeps the value that
1025 !! \c dH or \c dG holds on input, and its term moves to the right-hand side.
1026 !! Or both are fitted from exactly two potentials, whose two equations per
1027 !! row are solved exactly:
1028 !! \f[ H_{ii} = \frac{r_{1,i}\phi'_{2,i} - r_{2,i}\phi'_{1,i}}{D_i}, \quad
1029 !! G_{ii} = \frac{r_{1,i}\phi_{2,i} - r_{2,i}\phi_{1,i}}{D_i}, \quad
1030 !! D_i = \phi_{1,i}\phi'_{2,i} - \phi'_{1,i}\phi_{2,i} . \f]
1031 !! The ratio \f$s_i = |D_i| / (|\phi_{1,i}\phi'_{2,i}| +
1032 !! |\phi'_{1,i}\phi_{2,i}|)\f$, between 0 and 1, does not depend on how
1033 !! either potential is scaled; about \f$-\log_{10}s_i\f$ digits of
1034 !! \f$D_i\f$ are lost to cancellation. The routine stops with an error when
1035 !! \f$s_i < 10^{-6}\f$ at some row, where the two equations cannot separate
1036 !! \f$H_{ii}\f$ from \f$G_{ii}\f$, and warns when the smallest \f$s_i\f$ is
1037 !! below \f$10^{-2}\f$.
1038 integer function fit_diag(vac,phi,dphi,c_free,fit_H,fit_G,dH,dG) &
1039 &result(ierr)
1040 use vac_vars, only: in_context
1041
1042 character(*), parameter :: rout_name = 'fit_diag'
1043
1044 ! input / output
1045 type(vac_type), intent(in) :: vac !< vacuum variables
1046 real(dp), intent(in) :: phi(:,:) !< potentials at the boundary points, one column per potential
1047 real(dp), intent(in) :: dphi(:,:) !< their normal data
1048 real(dp), intent(in) :: c_free(:) !< free term of the identity of each potential
1049 logical, intent(in) :: fit_h !< fit the diagonal of H
1050 logical, intent(in) :: fit_g !< fit the diagonal of G
1051 real(dp), intent(inout) :: dh(:) !< diagonal of H (global): known on input, or fitted on output
1052 real(dp), intent(inout) :: dg(:) !< diagonal of G (global): known on input, or fitted on output
1053
1054 ! local variables
1055 integer :: kd ! counter
1056 integer :: rd_min ! row of the smallest s_i
1057 real(dp), parameter :: tol_stop = 1.e-6_dp ! smallest s_i accepted
1058 real(dp), parameter :: tol_warn = 1.e-2_dp ! smallest s_i without a warning
1059 real(dp), allocatable :: yh(:), yg(:) ! products with the full H and G
1060 real(dp), allocatable :: dh0(:), dg0(:) ! current diagonals of the matrices
1061 real(dp), allocatable :: rr(:,:) ! right-hand sides, per row and potential
1062 real(dp), allocatable :: det(:), s_sep(:) ! D_i and s_i
1063 character(len=max_str_ln) :: err_msg ! error message
1064
1065 ierr = 0
1066 if (.not.in_context(vac%ctxt_HG)) return
1067 if (size(c_free).ne.size(phi,2) .or. fit_h.and.fit_g.and.size(phi,2).ne.2) then
1068 ierr = 1
1069 err_msg = 'one free term per potential, and exactly two potentials &
1070 &to fit both diagonals'
1071 chckerr(err_msg)
1072 end if
1073 allocate(yh(vac%n_bnd),yg(vac%n_bnd),dh0(vac%n_bnd),dg0(vac%n_bnd))
1074 allocate(rr(vac%n_bnd,size(phi,2)))
1075 ierr = dis_diag(vac,vac%H,dh0)
1076 chckerr('')
1077 ierr = dis_diag(vac,vac%G,dg0)
1078 chckerr('')
1079 do kd = 1,size(phi,2)
1080 ierr = dis_matvec(vac,vac%H,vac%desc_H,phi(:,kd),yh)
1081 chckerr('')
1082 ierr = dis_matvec(vac,vac%G,vac%desc_G,dphi(:,kd),yg)
1083 chckerr('')
1084 rr(:,kd) = c_free(kd)*phi(:,kd) - (yh-dh0*phi(:,kd)) + &
1085 &(yg-dg0*dphi(:,kd))
1086 if (.not.fit_h) rr(:,kd) = rr(:,kd) - dh*phi(:,kd)
1087 if (.not.fit_g) rr(:,kd) = rr(:,kd) + dg*dphi(:,kd)
1088 end do
1089 if (fit_h .and. fit_g) then
1090 det = phi(:,1)*dphi(:,2) - dphi(:,1)*phi(:,2)
1091 s_sep = abs(phi(:,1)*dphi(:,2)) + abs(dphi(:,1)*phi(:,2))
1092 where (s_sep.gt.0._dp) s_sep = abs(det)/s_sep
1093 rd_min = minloc(s_sep,1)
1094 if (s_sep(rd_min).lt.tol_stop) then
1095 ierr = 1
1096 err_msg = 'row '//trim(i2str(rd_min))//' of '//&
1097 &trim(i2str(vac%n_bnd))//' (n = '//trim(i2str(vac%prim_X))//&
1098 &'): identities cannot separate H_ii and G_ii'
1099 chckerr(err_msg)
1100 end if
1101 if (s_sep(rd_min).lt.tol_warn) call writo('fit_diag: identities &
1102 &barely separate H_ii and G_ii at row '//trim(i2str(rd_min))//&
1103 &' (s = '//trim(r2strt(s_sep(rd_min)))//')',warning=.true.)
1104 dh = (rr(:,1)*dphi(:,2) - rr(:,2)*dphi(:,1))/det
1105 dg = (rr(:,1)*phi(:,2) - rr(:,2)*phi(:,1))/det
1106 else if (fit_h) then
1107 dh = sum(phi*rr,2)/sum(phi**2,2)
1108 else if (fit_g) then
1109 dg = -sum(dphi*rr,2)/sum(dphi**2,2)
1110 end if
1111 end function fit_diag
