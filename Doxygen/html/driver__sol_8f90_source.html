27 !> Main driver of PB3D solution part.
28 !!
29 !! - sets up:
30 !! * \c grid_sol (only first Richardson level)
31 !! * \c sol
32 !! - writes to HDF5:
33 !! * \c grid_sol (only first Richardson level)
34 !! * \c sol
35 !! - deallocates:
36 !! * sol before setting up (but after guess)
37 !!
38 !! \return ierr
39 integer function run_driver_sol(grid_eq,grid_X,grid_sol,X,vac,sol) &
40 &result(ierr)
41
42 use num_vars, only: ev_style, eq_style, rich_restart_lvl, rank, &
43 &n_procs, x_grid_style, jump_to_sol, bc_style
44 use grid_vars, only: n_r_sol, n_alpha
45 use pb3d_ops, only: reconstruct_pb3d_grid, reconstruct_pb3d_sol
46 use slepc_ops, only: solve_ev_system_slepc
47 use grid_ops, only: calc_norm_range, setup_grid_sol, &
48 &print_output_grid, redistribute_output_grid
49 use sol_ops, only: print_output_sol
50 use rich_vars, only: rich_lvl
51 use rich_ops, only: calc_rich_ex
52 use vac_ops, only: calc_vac_res, print_output_vac
53 use grid_utilities, only: trim_grid
54 use x_ops, only: setup_modes, redistribute_output_x
55 use sol_utilities, only: interp_v
56#if ldebug
57 use x_ops, only: print_debug_x_2
58#endif
59
60 character(*), parameter :: rout_name = 'run_driver_sol'
61
62 ! input / output
63 type(grid_type), intent(in) :: grid_eq !< equilibrium grid
64 type(grid_type), intent(in), target :: grid_x !< perturbation grid
65 type(grid_type), intent(inout) :: grid_sol !< solution grid
66 type(x_2_type), intent(in) :: x !< integrated tensorial perturbation variables
67 type(vac_type), intent(inout) :: vac !< vacuum variables
68 type(sol_type), intent(inout) :: sol !< solution variables
69
70 ! local variables
71 type(grid_type) :: grid_sol_trim ! trimmed solution grid
72 type(grid_type) :: grid_x_rdst ! redistributed perturbation grid
73 type(grid_type) :: grid_x_sol ! perturbation grid with solution normal part
74 type(x_2_type) :: x_rdst ! redistributed X
75 type(x_2_type) :: x_sol ! interpolated X
76 integer :: sol_limits(2) ! min. and max. index of sol grid for this process
77 integer :: rich_lvl_name ! either the Richardson level or zero, to append to names
78 real(dp), allocatable :: r_f_sol(:) ! normal points in solution grid
79 logical :: do_vac_ops ! whether specific calculations for vacuum are necessary
80 character(len=max_str_ln) :: err_msg ! error message
81
82 ! initialize ierr
83 ierr = 0
84
85 ! set up solution grid if first level
86 if (rich_lvl.eq.rich_restart_lvl) then
87 ! Divide solution grid under group processes, calculating the limits
88 ! and the normal coordinate.
89 allocate(r_f_sol(n_r_sol))
90 ierr = calc_norm_range('PB3D_sol',sol_limits=sol_limits,&
91 &r_f_sol=r_f_sol)
92 chckerr('')
93
94 if (rich_lvl.eq.1) then
95 call writo('Set up solution grid')
96 call lvl_ud(1)
97
98 call writo('Calculate the grid')
99 call lvl_ud(1)
100 ierr = setup_grid_sol(grid_eq,grid_x,grid_sol,r_f_sol,&
101 &sol_limits)
102 chckerr('')
103 call lvl_ud(-1)
104
105 call writo('Write to output file')
106 call lvl_ud(1)
107 ierr = print_output_grid(grid_sol,'solution','sol',&
108 &remove_previous_arrs=(jump_to_sol.and.&
109 &(x_grid_style.eq.1 .or. x_grid_style.eq.3)))
110 chckerr('')
111 call lvl_ud(-1)
112
113 ! set up modes
114 ierr = setup_modes(mds_sol,grid_eq,grid_sol,plot_name='sol')
115 chckerr('')
116
117 call lvl_ud(-1)
118 else
119 ! restore solution grid and trim it
120 ierr = reconstruct_pb3d_grid(grid_sol,'sol',&
121 &grid_limits=sol_limits,divided=n_procs.gt.1)
122 chckerr('')
123 ierr = trim_grid(grid_sol,grid_sol_trim)
124 chckerr('')
125
126 ! set up modes
127 ierr = setup_modes(mds_sol,grid_eq,grid_sol,plot_name='sol')
128 chckerr('')
129
130 ! reconstruct solution on trimmed grid
131 ierr = reconstruct_pb3d_sol(mds_sol,grid_sol_trim,sol,'sol',&
132 &rich_lvl=rich_lvl-1)
133 chckerr('')
134
135 ! clean up
136 call grid_sol_trim%dealloc()
137 end if
138
139 deallocate(r_f_sol)
140 end if
141
142 ! set up whether Richardson level has to be appended to the name and
143 ! whether to do vacuum operations
144 select case (eq_style)
145 case (1) ! VMEC
146 rich_lvl_name = rich_lvl ! append richardson level
147 do_vac_ops = .true.
148 case (2) ! HELENA
149 rich_lvl_name = 0 ! do not append name
150 if (rich_lvl.eq.rich_restart_lvl) then
151 do_vac_ops = .true.
152 else
153 do_vac_ops = .false.
154 end if
155 end select
156 if (bc_style(2).eq.1) do_vac_ops = .false. ! fixed-boundary BC does not use the vacuum response
157
158 if (do_vac_ops) then
159 ! calculate vacuum
160 ierr = calc_vac_res(mds_sol,vac)
161 chckerr('')
162
163 call writo('Write to output file')
164 call lvl_ud(1)
165 chckerr('')
166 if (rank.eq.n_procs-1) then
167 ierr = print_output_vac(vac,'vac',rich_lvl=rich_lvl_name)
168 chckerr('')
169 end if
170 call lvl_ud(-1)
171 end if
172
173 ! initialize interpolated X
174 ierr = grid_x_sol%init([1,n_alpha,grid_sol%n(3)],&
175 &[grid_sol%i_min,grid_sol%i_max],grid_sol%divided)
176 chckerr('')
177 grid_x_sol%r_F = grid_sol%r_F
178 grid_x_sol%r_E = grid_sol%r_E
179
180
181 select case (x_grid_style)
182 case (1,3) ! equilibrium or enriched
183 ! user output
184 call writo('Redistribute and interpolate the perturbation &
185 &variables to solution grid')
186 call lvl_ud(1)
187
188 ! redistribute grid and X variables
189 ierr = redistribute_output_grid(grid_x,grid_x_rdst)
190 chckerr('')
191 ierr = redistribute_output_x(mds_x,grid_x,grid_x_rdst,x,x_rdst)
192 chckerr('')
193
194 ! initialize
195 call x_sol%init(mds_sol,grid_x_sol,is_field_averaged=.true.)
196
197 ! interpolate
198 ierr = interp_v(mds_x,grid_x_rdst,x_rdst,mds_sol,grid_x_sol,&
199 &x_sol)
200 chckerr('')
201
202 ! clean up
203 call x_rdst%dealloc()
204 call grid_x_rdst%dealloc()
205
206 call lvl_ud(-1)
207 case (2) ! solution
208 ! user output
209 call writo('Copy the perturbation variables to solution grid')
210 call lvl_ud(1)
211
212 ! copy
213 call x%copy(mds_sol,grid_x,x_sol)
214
215 call lvl_ud(-1)
216 end select
217
218#if ldebug
219 ! write integrated field-aligned tensorial perturbation quantities
220 ! to output and plot
221 if (debug_run_driver_sol) then
222 ierr = print_debug_x_2(mds_sol,grid_x_sol,x_sol)
223 chckerr('')
224 end if
225#endif
226
227 ! clean up
228 call grid_x_sol%dealloc()
229
230 ! solve the system
231 call writo('Solving the system')
232 call lvl_ud(1)
233 select case (ev_style)
234 case(1) ! SLEPC solver for EV problem
235 ! solve the system
236 ierr = solve_ev_system_slepc(mds_sol,grid_sol,x_sol,vac,sol)
237 chckerr('')
238 case default
239 err_msg = 'No EV solver style associated with '//&
240 &trim(i2str(ev_style))
241 ierr = 1
242 chckerr(err_msg)
243 end select
244 call lvl_ud(-1)
245 call writo('System solved')
246
247 ! write solution variables to output
248 ierr = print_output_sol(grid_sol,sol,'sol',rich_lvl=rich_lvl,&
249 &remove_previous_arrs=&
250 &(jump_to_sol.and.(x_grid_style.eq.1 .or. x_grid_style.eq.3)))
251 chckerr('')
252
253 ! calculate Richardson extrapolation factors if necessary
254 call calc_rich_ex(sol%val)
255
256 ! clean up
257 call x_sol%dealloc()
258 end function run_driver_sol
259end module driver_sol
