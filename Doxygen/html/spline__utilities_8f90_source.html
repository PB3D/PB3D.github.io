37 !> \private real version
38 !!
39 !! One point gives the constant through it, so its derivatives are zero;
40 !! no point is an error.
41 integer function spline_real(x,y,xnew,ynew,ord,deriv,bcs,bcs_val,extrap) &
42 &result(ierr)
43
44 use ezspline_obj
45 use ezspline
46
47 character(*), parameter :: rout_name = 'spline_real'
48
49 ! input / output
50 real(dp), intent(in), target :: x(:) !< coordinates
51 real(dp), intent(in) :: y(:) !< function value
52 real(dp), intent(in), target :: xnew(:) !< new coordinates
53 real(dp), intent(out) :: ynew(:) !< new function values
54 integer, intent(in), optional :: ord !< order [def 3]
55 integer, intent(in), optional :: deriv !< derivative [def 0]
56 integer, intent(in), optional :: bcs(2) !< boundary conditions [def 0]
57 real(dp), intent(in), optional :: bcs_val(2) !< boundary conditions [no def]
58 logical, intent(in), optional :: extrap !< whether extrapolation is allowed [def .false.]
59
60 ! local variables
61 type(ezspline1) :: f_spl ! spline object
62 integer :: ord_loc ! local order
63 integer :: deriv_loc ! local deriv
64 integer :: bcs_loc(2) ! local bcs
65 integer :: kd ! counter
66 integer :: n ! size of x, y
67 integer :: nnew ! size of output x, y
68 integer :: nnew_interp ! size of interpolated x, y, rest is extrapolated
69 integer :: il(2) ! limits of interp., what falls outside is extrapolated
70 real(dp), pointer :: x_loc(:) ! x or -x
71 real(dp), pointer :: xnew_loc(:) ! xnew or -xnew
72 real(dp) :: bcs_val_loc(2) ! local bcs_val
73 real(dp) :: lim_vals(0:3) ! limit values at boundaries for extrapolation
74 real(dp), allocatable :: xnew_ez(:) ! xnew in EZ spline doubles
75 real(dp), allocatable :: ynew_ez(:) ! ynew in EZ spline doubles
76 character(len=max_str_ln) :: err_msg ! error message
77 logical :: flip_x ! whether x axis is flipped
78 logical :: extrap_loc ! local extrap
79
80 ! initialize ierr
81 ierr = 0
82
83 ! set local variables
84 ord_loc = 3
85 if (present(ord)) ord_loc = ord
86 deriv_loc = 0
87 if (present(deriv)) deriv_loc = deriv
88 bcs_loc = [0,0]
89 if (present(bcs)) bcs_loc = bcs
90 if (present(bcs_val)) bcs_val_loc = bcs_val
91 extrap_loc = .false.
92 if (present(extrap)) extrap_loc = extrap
93
94 ! set other variables
95 n = size(x)
96 nnew = size(xnew)
97 if (n.lt.1) then
98 ierr = 1
99 chckerr('Need at least one point')
100 end if
101
102 ! set local x and xnew, possibly flipped if negative
103 flip_x = .false.
104 if (n.gt.1) flip_x = x(2).lt.x(1)
105 if (flip_x) then
106 allocate(x_loc(n))
107 allocate(xnew_loc(nnew))
108 x_loc = -x
109 xnew_loc = -xnew
110 else
111 x_loc => x
112 xnew_loc => xnew
113 end if
114
115#if ldebug
116 ! check monotony
117 do kd = 2,n
118 if (x_loc(kd).le.x_loc(kd-1)) then
119 ierr = 1
120 err_msg = '|x| is not monotonously increasing'
121 chckerr(err_msg)
122 end if
123 end do
124
125 ! check array sizes
126 if (size(y).ne.n) then
127 ierr = 1
128 err_msg = 'x and y need to have the same size'
129 chckerr(err_msg)
130 end if
131 if (size(ynew).ne.nnew) then
132 ierr = 1
133 err_msg = 'xnew and ynew need to have the same size'
134 chckerr(err_msg)
135 end if
136
137 ! check order
138 if (ord_loc.lt.1 .or. ord_loc.gt.3) then
139 ierr = 1
140 err_msg = 'Only order 1 (linear), 2 (akima hermite) or 3 (cubic) &
141 &possible'
142 chckerr(err_msg)
143 end if
144
145 ! check derviative
146 select case (deriv_loc)
147 case (:-1)
148 ierr = 1
149 err_msg = 'only nonnegative degrees of derivative are possible'
150 chckerr(err_msg)
151 case (0:1)
152 ! do nothing
153 case (2:3)
154 if (ord_loc.eq.1 .or. ord_loc.eq.2) then
155 ierr = 1
156 err_msg = 'Derivative of degree '//trim(i2str(deriv_loc))//&
157 &' not possible for order '//trim(i2str(ord_loc))
158 chckerr(err_msg)
159 end if
160 case (4:)
161 ierr = 1
162 err_msg = 'maximum degree of derivative is 2 for order 4'
163 chckerr(err_msg)
164 end select
165
166 ! check boundary conditions
167 select case (ord_loc)
168 case (1)
169 if (present(bcs) .or. present(bcs_val)) then
170 call writo('for order 1, no boundary conditions can be &
171 &prescribed',alert=.true.)
172 end if
173 case (2:3)
174 ! check possibility
175 do kd = 1,2
176 if (bcs_loc(kd).lt.-1 .or. bcs_loc(kd).gt.ord_loc-1) then
177 ierr = 1
178 err_msg = 'For order '//trim(i2str(ord_loc))//&
179 &' bcs has to be -1..'//trim(i2str(ord_loc-1))
180 chckerr(err_msg)
181 end if
182 end do
183
184 ! check whether derivatives are provided
185 if ((any(bcs_loc.eq.1) .or. any(bcs_loc.eq.2)) &
186 &.and. .not.present(bcs_val)) then
187 ierr = 1
188 err_msg = 'When prescribing first or second derviatives, &
189 &need to provide bcs_val'
190 chckerr(err_msg)
191 end if
192 end select
193#endif
194
195 ! set up interpolation limits
196 do kd = 1,nnew
197 if (xnew_loc(kd).ge.x_loc(1)) exit
198 end do
199 il(1) = kd
200 do kd = nnew,1,-1
201 if (xnew_loc(kd).le.x_loc(n)) exit
202 end do
203 il(2) = kd
204 nnew_interp = il(2)-il(1)+1
205
206 ! check for extrapolation
207 if ((il(1).ne.1 .or. il(2).ne.nnew) .and. &
208 &.not.extrap_loc) then
209 ierr = 1
210 call writo('xnew = ['//trim(r2str(minval(xnew)))//'..'//&
211 &trim(r2str(maxval(xnew)))//']')
212 call writo('but x = ['//trim(r2str(minval(x)))//'..'//&
213 &trim(r2str(maxval(x)))//']')
214 err_msg = 'Extrapolation needed, but not allowed'
215 chckerr(err_msg)
216 end if
217
218 ! one point: the constant through it
219 if (n.eq.1) then
220 ynew = merge(y(1),0._dp,deriv_loc.eq.0)
221 return
222 end if
223
224 ! initialize
225 if (ord_loc.eq.1) then
226 call ezlinear_init(f_spl,n,ierr)
227 call ezspline_error(ierr)
228 chckerr('')
229 else
230 call ezspline_init(f_spl,n,bcs_loc,ierr)
231 call ezspline_error(ierr)
232 chckerr('')
233 if (ord_loc.eq.2) f_spl%isHermite = 1
234 end if
235
236 ! set grid
237 f_spl%x1 = x_loc
238
239 ! set boundary condition
240 if (present(bcs_val)) then
241 f_spl%bcval1min = bcs_val_loc(1)
242 f_spl%bcval1max = bcs_val_loc(2)
243 if (flip_x) then ! if x-axis flipped, also first derivative flipped
244 if (bcs_loc(1) .eq. 1) f_spl%bcval1min = -f_spl%bcval1min
245 if (bcs_loc(2) .eq. 1) f_spl%bcval1max = -f_spl%bcval1max
246 end if
247 end if
248
249 ! set up
250 call ezspline_setup(f_spl,y,ierr,exact_dim=.true.) ! match exact dimensions
251 call ezspline_error(ierr)
252 chckerr('')
253 ierr = keep_knots(f_spl%x1,f_spl%x1pkg,f_spl%klookup1)
254 chckerr('')
255
256 ! interpolated part
257 if (il(1).le.il(2)) then
258 allocate(xnew_ez(nnew_interp))
259 allocate(ynew_ez(nnew_interp))
260 xnew_ez = xnew_loc(il(1):il(2))
261
262 if (deriv_loc.eq.0) then
263 ! interpolate
264 call ezspline_interp(f_spl,nnew_interp,xnew_ez,ynew_ez,ierr)
265 call ezspline_error(ierr)
266 chckerr('')
267 else
268 call ezspline_derivative(f_spl,deriv_loc,nnew_interp,xnew_ez,&
269 &ynew_ez,ierr)
270 call ezspline_error(ierr)
271 chckerr('')
272 end if
273
274 ynew(il(1):il(2)) = ynew_ez
275 deallocate(xnew_ez,ynew_ez)
276 end if
277
278 ! extrapolated part
279 if (extrap_loc) then
280 ! extrapolate left
281 if (il(1).gt.1) then
282 ! set up limit values at first point
283 ierr = setup_lim_vals(x_loc(1),lim_vals)
284 chckerr('')
285
286 ! calculate extrapolation
287 call calc_extrap(xnew_loc(1:il(1)-1),x_loc(1),lim_vals,&
288 &ynew(1:il(1)-1))
289 end if
290
291 ! extrapolate right
292 if (il(2).lt.nnew) then
293 ! set up limit values at last point
294 ierr = setup_lim_vals(x_loc(n),lim_vals)
295 chckerr('')
296
297 ! calculate extrapolation
298 call calc_extrap(xnew_loc(il(2)+1:nnew),x_loc(n),lim_vals,&
299 &ynew(il(2)+1:nnew))
300 end if
301 end if
302
303 ! with the x axis flipped, the odd derivatives are with respect to -x
304 if (flip_x .and. mod(deriv_loc,2).eq.1) ynew = -ynew
305
306 ! free
307 call ezspline_free(f_spl,ierr)
308 chckerr('')
309 call ezspline_error(ierr)
310 chckerr('')
311 if (flip_x) then
312 deallocate(x_loc)
313 deallocate(xnew_loc)
314 end if
315 nullify(x_loc)
316 nullify(xnew_loc)

357 !> /private calculate extrapolation
358 subroutine calc_extrap(xnew,xb,lim_vals,ynew)
359 ! input / output
360 real(dp), intent(in) :: xnew(:) ! xnew where to extrapolate
361 real(dp), intent(in) :: xb ! x of boundary
362 real(dp), intent(in) :: lim_vals(0:2) ! limit values
363 real(dp), intent(out) :: ynew(:) ! y at xnew
364
365 ! local variables
366 real(dp), allocatable :: xdel(:) ! delta x for extrapolation
367
368 allocate(xdel(size(xnew)))
369 xdel = xnew-xb
370
371 select case (deriv_loc)
372 case (0)
373 ynew = lim_vals(0) + &
374 &lim_vals(1)*xdel + &
375 &lim_vals(2)*xdel**2*0.5_dp
376 case (1)
377 ynew = lim_vals(1) + &
378 &lim_vals(2)*xdel
379 case (2)
380 ynew = lim_vals(2)
381 end select
382 end subroutine calc_extrap
383 end function spline_real

588 !> \private make PSPLINE look up the points of an evaluation on the knots
589 !!
590 !! \c EZspline_setup lets PSPLINE's \c genxpkg treat the grid x as evenly
591 !! spaced when every spacing lies within 5e-7 max|x| of the mean spacing.
592 !! The evaluation at an array of points then finds the cell and the
593 !! position in it on the evenly spaced grid, while the coefficients of
594 !! the spline belong to the knots x, so that at a knot the spline misses
595 !! its value by its slope times the distance between the knot and the
596 !! point of the evenly spaced grid. On a short range of the poloidal flux
597 !! where the rotational transform hardly changes, this is 1e-6 and more
598 !! (issue #95).
599 !!
600 !! This routine sets up the lookup table \c xpkg again with a tolerance
601 !! of 1e-13 max|x|, so that only a grid that is even to rounding is still
602 !! treated as evenly spaced.
603 !!
604 !! \return ierr
605 integer function keep_knots(x,xpkg,klookup) result(ierr)
606 character(*), parameter :: rout_name = 'keep_knots'
607
608 ! input / output
609 real(dp), intent(in) :: x(:) !< knots
610 real(dp), intent(inout) :: xpkg(size(x),4) !< lookup table of PSPLINE for x
611 integer, intent(in) :: klookup !< lookup method of PSPLINE for an uneven grid
612
613 ! local variables
614 real(dp), parameter :: tol_even = 1.e-13_dp ! tolerance of an even grid, relative to max|x|
615
616 interface
617 subroutine genxpkg(nx,x,xpkg,iper,imsg,itol,ztol,ialg,ier)
618 import dp
619 integer :: nx, iper, imsg, itol, ialg, ier
620 real(dp) :: x(nx), xpkg(nx,4), ztol
621 end subroutine genxpkg
622 end interface

624 ! initialize ierr
625 ierr = 0
